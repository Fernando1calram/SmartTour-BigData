# 13

Las tres relaciones son de uno a muchos:
    * Municipios -> Hoteles: un municipio puede tener 0, uno o mucho y cada hotel pertenece a un municipio
    * Municipios -> Monumentos: un municipio puede tener cero, uno o muchos monumentos y cada monumento pertenece a un municipio
    * Municipios -> Oficinas de turismo: un municipio puede tener cero, una o muchas oficinas y cada ofricina pertenece a un municipio

- ¿Qué oficinas tienen una mayor afluencia durante los fines de semana?

SELECT
    o.nombre AS oficina,
    SUM(v.num_visitas) FILTER (
        WHERE f.es_fin_semana = TRUE
    ) AS visitas_fin_de_semana,
    SUM(v.num_visitas) FILTER (
        WHERE f.es_fin_semana = FALSE
    ) AS visitas_dias_laborables,
    ROUND(
        SUM(v.num_visitas) FILTER (
            WHERE f.es_fin_semana = TRUE
        )::NUMERIC
        /
        NULLIF(
            SUM(v.num_visitas) FILTER (
                WHERE f.es_fin_semana = FALSE
            ),
            0
        ),
        2
    ) AS ratio_fin_semana_laborables
FROM visitadadas v
JOIN dim_oficina_turismo o
    ON v.oficina_key = o.oficina_key
JOIN fechas f
    ON v.fecha_key = f.fecha_key
GROUP BY
    o.oficina_key,
    o.nombre
ORDER BY
    visitas_fin_de_semana DESC;

- ¿Qué monumentos tienen mayor estacionalidad?

WITH visitas_mensuales AS (
    SELECT
        mo.monumento_key,
        mo.nombre AS monumento,
        f.mes,
        f.nombre_mes,
        SUM(v.num_visitas) AS visitas_mes
    FROM visitas v
    JOIN dim_monumento mo
        ON v.monumento_key = mo.monumento_key
    JOIN fechas f
        ON v.fecha_key = f.fecha_key
    GROUP BY
        mo.monumento_key,
        mo.nombre,
        f.mes,
        f.nombre_mes
),
media_monumento AS (
    SELECT
        monumento_key,
        AVG(visitas_mes) AS media_mensual
    FROM visitas_mensuales
    GROUP BY monumento_key
)
SELECT
    vm.monumento,
    vm.nombre_mes,
    vm.visitas_mes,
    ROUND(mm.media_mensual, 2) AS media_mensual,
    ROUND(
        (vm.visitas_mes - mm.media_mensual) / NULLIF(mm.media_mensual, 0) * 100,
        2
    ) AS desviacion_porcentual
FROM visitas_mensuales vm
JOIN media_monumento mm
    ON vm.monumento_key = mm.monumento_key
ORDER BY
    desviacion_porcentual DESC,
    vm.monumento,
    vm.mes;

# 4

- Esa vista permitirá comparar, para cada municipio y periodo:
- Turistas alojados procedentes de reservas.
- Visitas a oficinas procedentes de visitadadas.
- Visitas a monumentos procedentes de visitas.

CREATE OR REPLACE VIEW vw_actividad_turistica_municipio AS
WITH reservas_mensuales AS (
    SELECT
        r.municipio_key,
        f.anio,
        f.mes,
        f.nombre_mes,
        SUM(r.turistas) AS turistas_alojados,
        SUM(r.importe) AS ingresos_hoteles,
        COUNT(*) AS num_reservas
    FROM reservas r
    JOIN fechas f
        ON r.fecha_entrada_key = f.fecha_key
    GROUP BY
        r.municipio_key,
        f.anio,
        f.mes,
        f.nombre_mes
),
oficinas_mensuales AS (
    SELECT
        v.municipio_key,
        f.anio,
        f.mes,
        f.nombre_mes,
        SUM(v.num_visitas) AS visitas_oficinas,
        SUM(v.num_consultas) AS consultas_turisticas
    FROM visitadadas v
    JOIN fechas f
        ON v.fecha_key = f.fecha_key
    GROUP BY
        v.municipio_key,
        f.anio,
        f.mes,
        f.nombre_mes
),
monumentos_mensuales AS (
    SELECT
        v.municipio_key,
        f.anio,
        f.mes,
        f.nombre_mes,
        SUM(v.num_visitas) AS visitas_monumentos,
        SUM(v.num_entradas) AS entradas_monumentos,
        SUM(v.importe_entradas) AS ingresos_monumentos
    FROM visitas v
    JOIN fechas f
        ON v.fecha_key = f.fecha_key
    GROUP BY
        v.municipio_key,
        f.anio,
        f.mes,
        f.nombre_mes
)
SELECT
    mu.nombre AS municipio,
    mu.provincia,
    r.anio,
    r.mes,
    r.nombre_mes,
    COALESCE(r.turistas_alojados, 0) AS turistas_alojados,
    COALESCE(r.ingresos_hoteles, 0) AS ingresos_hoteles,
    COALESCE(r.num_reservas, 0) AS num_reservas,
    COALESCE(o.visitas_oficinas, 0) AS visitas_oficinas,
    COALESCE(o.consultas_turisticas, 0) AS consultas_turisticas,
    COALESCE(m.visitas_monumentos, 0) AS visitas_monumentos,
    COALESCE(m.entradas_monumentos, 0) AS entradas_monumentos,
    COALESCE(m.ingresos_monumentos, 0) AS ingresos_monumentos
FROM reservas_mensuales r
LEFT JOIN oficinas_mensuales o
    ON r.municipio_key = o.municipio_key
   AND r.anio = o.anio
   AND r.mes = o.mes
LEFT JOIN monumentos_mensuales m
    ON r.municipio_key = m.municipio_key
   AND r.anio = m.anio
   AND r.mes = m.mes
JOIN dim_municipio mu
    ON r.municipio_key = mu.municipio_key;

CREATE OR REPLACE VIEW public.vw_actividad_turistica_municipio AS

WITH reservas_mes AS (
    SELECT
        r.municipio_key,
        f.anio,
        f.mes,
        COUNT(*) AS total_reservas,
        SUM(r.turistas) AS turistas_en_reservas,
        SUM(r.importe) AS importe_reservas
    FROM public.reservas r
    JOIN public.fechas f
        ON r.fecha_entrada_key = f.fecha_key
    GROUP BY
        r.municipio_key,
        f.anio,
        f.mes
),

oficinas_mes AS (
    SELECT
        v.municipio_key,
        f.anio,
        f.mes,
        SUM(v.num_visitas) AS visitas_oficinas
    FROM public.visitadadas v
    JOIN public.fechas f
        ON v.fecha_key = f.fecha_key
    GROUP BY
        v.municipio_key,
        f.anio,
        f.mes
),

monumentos_mes AS (
    SELECT
        v.municipio_key,
        f.anio,
        f.mes,
        SUM(v.num_visitas) AS visitas_monumentos,
        SUM(v.importe_entradas) AS ingresos_entradas
    FROM public.visitas v
    JOIN public.fechas f
        ON v.fecha_key = f.fecha_key
    GROUP BY
        v.municipio_key,
        f.anio,
        f.mes
),

periodos AS (
    SELECT municipio_key, anio, mes
    FROM reservas_mes

    UNION

    SELECT municipio_key, anio, mes
    FROM oficinas_mes

    UNION

    SELECT municipio_key, anio, mes
    FROM monumentos_mes
)

SELECT
    m.municipio_key,
    m.nombre AS municipio,
    m.provincia,
    p.anio,
    p.mes,
    COALESCE(r.total_reservas, 0) AS total_reservas,
    COALESCE(r.turistas_en_reservas, 0) AS turistas_en_reservas,
    COALESCE(r.importe_reservas, 0) AS importe_reservas,
    COALESCE(o.visitas_oficinas, 0) AS visitas_oficinas,
    COALESCE(mo.visitas_monumentos, 0) AS visitas_monumentos,
    COALESCE(mo.ingresos_entradas, 0) AS ingresos_entradas
FROM periodos p
JOIN public.dim_municipio m
    ON p.municipio_key = m.municipio_key

LEFT JOIN reservas_mes r
    ON p.municipio_key = r.municipio_key
   AND p.anio = r.anio
   AND p.mes = r.mes

LEFT JOIN oficinas_mes o
    ON p.municipio_key = o.municipio_key
   AND p.anio = o.anio
   AND p.mes = o.mes

LEFT JOIN monumentos_mes mo
    ON p.municipio_key = mo.municipio_key
   AND p.anio = mo.anio
   AND p.mes = mo.mes;

SELECT *

ORDER BY

    mes,
    turistas_en_reservas DESC,
